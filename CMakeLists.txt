cmake_minimum_required(VERSION 3.25)
project(VibeCraft VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VC_RENDERER "Build Vulkan test bed" ON)
option(VC_GPU_TESTS "Include offscreen Vulkan validation smoke test in CTest" OFF)
option(VC_SANITIZERS "Enable ASan and UBSan" OFF)
find_package(Threads REQUIRED)
add_library(vc_core engine/core/world.cpp engine/core/tasks.cpp engine/core/player.cpp engine/core/terrain_lod.cpp engine/core/caves.cpp)
target_include_directories(vc_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/engine)
target_link_libraries(vc_core PUBLIC Threads::Threads)
if(MSVC)
 target_compile_options(vc_core PUBLIC /W4 /permissive-)
else()
 target_compile_options(vc_core PUBLIC -Wall -Wextra -Wpedantic)
 if(VC_SANITIZERS)
  target_compile_options(vc_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vc_core PUBLIC -fsanitize=address,undefined)
 endif()
endif()
add_executable(vc_tests tests/tests.cpp)
target_link_libraries(vc_tests PRIVATE vc_core)
add_executable(vc_bench benchmarks/bench.cpp)
target_link_libraries(vc_bench PRIVATE vc_core)
enable_testing()
add_test(NAME core COMMAND vc_tests)
add_test(NAME streaming_smoke COMMAND vc_bench --frames 60 --scenario teleport --output ${CMAKE_BINARY_DIR}/smoke)
if(VC_RENDERER)
 find_package(Vulkan 1.3 REQUIRED)
 find_package(glfw3 3.3 REQUIRED)
 find_program(DXC dxc)
 if(NOT DXC)
  find_program(GLSLANG glslangValidator REQUIRED)
 endif()
 foreach(STAGE vert frag)
  if(STAGE STREQUAL "vert")
   set(PROFILE vs_6_0)
  else()
   set(PROFILE ps_6_0)
  endif()
  set(OUT ${CMAKE_CURRENT_BINARY_DIR}/shaders/terrain.${STAGE}.spv)
  if(DXC)
   add_custom_command(OUTPUT ${OUT} COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders COMMAND ${DXC} -spirv -fspv-target-env=vulkan1.3 -T ${PROFILE} -E main -Fo ${OUT} ${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/terrain.${STAGE}.hlsl DEPENDS assets/shaders/terrain.${STAGE}.hlsl VERBATIM)
  else()
   add_custom_command(OUTPUT ${OUT} COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders COMMAND ${GLSLANG} -D -V --target-env vulkan1.3 -S ${STAGE} -e main -o ${OUT} ${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/terrain.${STAGE}.hlsl DEPENDS assets/shaders/terrain.${STAGE}.hlsl VERBATIM)
  endif()
  list(APPEND SHADERS ${OUT})
 endforeach()
 add_custom_target(vc_shaders DEPENDS ${SHADERS})
 add_executable(vibecraft game/main.cpp game/play.cpp engine/render/vulkan/renderer.cpp)
 target_link_libraries(vibecraft PRIVATE vc_core Vulkan::Vulkan glfw)
 target_compile_definitions(vibecraft PRIVATE VC_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders")
 add_dependencies(vibecraft vc_shaders)
 if(NOT MSVC)
  target_compile_options(vibecraft PRIVATE -Wno-missing-field-initializers)
 endif()
 if(VC_GPU_TESTS)
  add_test(NAME vulkan_smoke COMMAND vibecraft --headless --validation --frames 12 --width 320 --height 180 --radius 2 --screenshot ${CMAKE_BINARY_DIR}/smoke.ppm --output ${CMAKE_BINARY_DIR}/render-smoke)
  set_tests_properties(vulkan_smoke PROPERTIES TIMEOUT 60)
 endif()
endif()
